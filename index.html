[...s.periods].reverse().find(p => d >= new Date(p.startDate)) : null; if (!activeP) return; const totalWeeks = Math.round((d - new Date(activeP.startDate)) / 604800000); let h = 0; for (let j = 0; j <= i; j++) if (skipped(j) && thursdays[j] >= new Date(activeP.startDate)) h++; const eff = totalWeeks - h; if (eff >= 0 && eff % activeP.cycle === 0) payIdx.push({ i, price: Number(activeP.price || 0), cycle: Number(activeP.cycle || 4) });
