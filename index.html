// دالة لتفعيل ميزة السحب والإفلات على المواعيد function enableDragAndDrop() { // تحديد جميع المواعيد (slots) وحاويات الأيام (slots containers) const draggables = document.querySelectorAll('.slot'); const containers = document.querySelectorAll('.slots'); draggables.forEach(draggable => { // جعل العنصر قابلاً للسحب draggable.setAttribute('draggable', 'true'); // عند بدء السحب draggable.addEventListener('dragstart', () => { draggable.classList.add('dragging'); }); // عند انتهاء السحب draggable.addEventListener('dragend', () => { draggable.classList.remove('dragging'); // يمكنك هنا استدعاء دالة لتحديث قاعدة البيانات (db) بالترتيب الجديد // commitChange(); }); }); containers.forEach(container => { // عند مرور العنصر المسحوب فوق حاوية اليوم container.addEventListener('dragover', e => { e.preventDefault(); // ضروري للسماح بالإفلات // تحديد العنصر الذي سيتم الإفلات قبله بناءً على موقع الماوس أفقياً (X-axis) const afterElement = getDragAfterElement(container, e.clientX); const draggable = document.querySelector('.dragging'); if (afterElement == null) { container.appendChild(draggable); // وضعه في النهاية إذا لم يكن هناك عنصر بعده } else { container.insertBefore(draggable, afterElement); // وضعه قبل العنصر المحسوب } }); }); } // دالة مساعدة لحساب الموقع الأفقي الدقيق (للسحب بالعرض) function getDragAfterElement(container, x) { // الحصول على جميع المواعيد في الحاوية باستثناء العنصر الذي يتم سحبه حالياً const draggableElements = [...container.querySelectorAll('.slot:not(.dragging)')]; return draggableElements.reduce((closest, child) => { const box = child.getBoundingClientRect(); // حساب المسافة بين منتصف العنصر ومؤشر الماوس أفقياً const offset = x - box.left - box.width / 2; if (offset < 0 && offset > closest.offset) { return { offset: offset, element: child }; } else { return closest; } }, { offset: Number.NEGATIVE_INFINITY }).element; }
